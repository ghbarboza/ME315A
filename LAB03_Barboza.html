getStatsAtraso <- function(input, pos) {
  input %>%
    filter(
      !grepl("^1", DESTINATION_AIRPORT),
      !is.na(DESTINATION_AIRPORT),
      !is.na(ARRIVAL_DELAY)
    ) %>%
    group_by(DESTINATION_AIRPORT) %>%
    summarise(
      soma = sum(ARRIVAL_DELAY),
      n = n(),
      .groups = "drop"
    )
}

flights <- read_csv_chunked(
  "flights.csv",
  DataFrameCallback$new(getStatsAtraso),
  chunk_size = 1000000,
  col_types = cols_only(
    DESTINATION_AIRPORT = col_character(),
    ARRIVAL_DELAY = col_double()
  )
) %>%
  group_by(DESTINATION_AIRPORT) %>%
  summarise(
    ARRIVAL_DELAY = sum(soma) / sum(n),
    .groups = "drop"
  )

A chave na tabela airports é IATA_CODE.
